import os from pydantic import BaseModel, Field from typing import List from reportlab.lib.pagesizes import A4 from reportlab.lib import colors from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle # ========================================== # 1. MODÈLES DE DONNÉES (Pydantic) # ========================================== class LigneFacture(BaseModel): description: str quantite: float prix_unitaire_ht: float total_ht: float class ClientBTP(BaseModel): id_client: str nom_entreprise: str nom_contact: str email: str telephone: str adresse: str nom_chantier: str lignes_prestation: List[LigneFacture] retenue_garantie_pct: float = 5.0 autoliquidation_tva: bool = False taux_tva: float = 20.0 # ========================================== # 2. FONCTION DE GÉNÉRATION DE PDF PAR CLIENT # ========================================== def generer_pdf_client(client: ClientBTP, dossier_sortie: str = "pdf_clients"): """Génère un PDF professionnel complet pour un client spécifique.""" # Créer le dossier s'il n'existe pas os.makedirs(dossier_sortie, exist_ok=True) nom_fichier = os.path.join(dossier_sortie, f"Fiche_Client_{client.id_client}_{client.nom_entreprise.replace(' ', '_')}.pdf") doc = SimpleDocTemplate( nom_fichier, pagesize=A4, rightMargin=30, leftMargin=30, topMargin=30, bottomMargin=30 ) styles = getSampleStyleSheet() # Custom styles style_titre = ParagraphStyle('Titre', parent=styles['Heading1'], fontName='Helvetica-Bold', fontSize=18, textColor=colors.HexColor('#1E3A8A'), spaceAfter=10) style_section = ParagraphStyle('Section', parent=styles['Heading2'], fontName='Helvetica-Bold', fontSize=12, textColor=colors.HexColor('#2563EB'), spaceBefore=10, spaceAfter=8) style_texte = ParagraphStyle('Texte', parent=styles['Normal'], fontName='Helvetica', fontSize=10, leading=14) elements = [] # --- En-tête --- elements.append(Paragraph(f"Dossier Client BTP & Synthèse Comptable", style_titre)) elements.append(Spacer(1, 10)) # --- Section Informations Client & Chantier --- elements.append(Paragraph("1. Informations Client & Chantier", style_section)) data_info = [ [Paragraph("Entreprise / Client :", style_texte), Paragraph(client.nom_entreprise, style_texte)], [Paragraph("Contact Principal :", style_texte), Paragraph(client.nom_contact, style_texte)], [Paragraph("Email / Téléphone :", style_texte), Paragraph(f"{client.email} | {client.telephone}", style_texte)], [Paragraph("Adresse :", style_texte), Paragraph(client.adresse, style_texte)], [Paragraph("Chantier Associe :", style_texte), Paragraph(f"{client.nom_chantier}", style_texte)], ] t_info = Table(data_info, colWidths=[150, 380]) t_info.setStyle(TableStyle([ ('BACKGROUND', (0,0), (-1,-1), colors.HexColor('#F8FAFC')), ('VALIGN', (0,0), (-1,-1), 'MIDDLE'), ('PADDING', (0,0), (-1,-1), 6), ('GRID', (0,0), (-1,-1), 0.5, colors.HexColor('#E2E8F0')), ])) elements.append(t_info) elements.append(Spacer(1, 15)) # --- Section Détail des Prestations / Travaux --- elements.append(Paragraph("2. Détail des Travaux & Prestations", style_section)) table_travaux = [["Description des travaux", "Qté", "P.U. HT (€)", "Total HT (€)"]] total_ht_brut = 0.0 for ligne in client.lignes_prestation: table_travaux.append([ Paragraph(ligne.description, style_texte), f"{ligne.quantite:.1f}", f"{ligne.prix_unitaire_ht:.2f} €", f"{ligne.total_ht:.2f} €" ]) total_ht_brut += ligne.total_ht t_travaux = Table(table_travaux, colWidths=[260, 50, 110, 110]) t_travaux.setStyle(TableStyle([ ('BACKGROUND', (0,0), (-1,0), colors.HexColor('#1E3A8A')), ('TEXTCOLOR', (0,0), (-1,0), colors.white), ('ALIGN', (1,0), (-1,-1), 'RIGHT'), ('FONTNAME', (0,0), (-1,0), 'Helvetica-Bold'), ('GRID', (0,0), (-1,-1), 0.5, colors.HexColor('#CBD5E1')), ('PADDING', (0,0), (-1,-1), 6), ])) elements.append(t_travaux) elements.append(Spacer(1, 15)) # --- Section Synthèse Comptable BTP --- elements.append(Paragraph("3. Synthèse Comptable (Spécifique BTP)", style_section)) # Calculs BTP montant_retenue = total_ht_brut * (client.retenue_garantie_pct / 100.0) ht_apres_retenue = total_ht_brut - montant_retenue if client.autoliquidation_tva: montant_tva = 0.0 libelle_tva = "0.00 € (Autoliquidation)" else: montant_tva = ht_apres_retenue * (client.taux_tva / 100.0) libelle_tva = f"{montant_tva:.2f} € ({client.taux_tva}%)" net_a_payer = ht_apres_retenue + montant_tva data_compta = [ ["Total Brut HT des travaux", f"{total_ht_brut:.2f} €"], [f"Retenue de garantie (-{client.retenue_garantie_pct:.1f}%)", f"- {montant_retenue:.2f} €"], ["Total HT Net (après retenue)", f"{ht_apres_retenue:.2f} €"], ["Montant TVA", libelle_tva], ["Net à Payer TTC", f"{net_a_payer:.2f} €"] ] t_compta = Table(data_compta, colWidths=[310, 220]) t_compta.setStyle(TableStyle([ ('ALIGN', (1,0), (1,-1), 'RIGHT'), ('FONTNAME', (0,-1), (-1,-1), 'Helvetica-Bold'), ('BACKGROUND', (0,-1), (-1,-1), colors.HexColor('#FEF08A')), # Highlight du net à payer ('GRID', (0,0), (-1,-1), 0.5, colors.HexColor('#94A3B8')), ('PADDING', (0,0), (-1,-1), 6), ])) elements.append(t_compta) # Mention légale Autoliquidation si nécessaire if client.autoliquidation_tva: elements.append(Spacer(1, 10)) elements.append(Paragraph( "Note comptable : Régime d'autoliquidation de la TVA applicable (Art. 283-2 nonies du CGI). Autoliquidation par le preneur.", styles['Italic'] )) doc.build(elements) print(f"✅ PDF généré pour client : {client.nom_entreprise} -> {nom_fichier}") # ========================================== # 3. EXÉCUTION EN BOUCLE POUR TOUS LES CLIENTS # ========================================== if __name__ == "__main__": # Liste de clients (données réelles ou extraites via votre IA/LLM) liste_clients = [ ClientBTP( id_client="CL-001", nom_entreprise="SCI Les Lilas", nom_contact="Jean Dupont", email="j.dupont@lilas.fr", telephone="06 12 34 56 78", adresse="12 Rue de la Paix, 75002 Paris", nom_chantier="Rénovation Siège Social - Lot Plâtrerie", lignes_prestation=[ LigneFacture(description="Pose de cloisons Placoplatre BA13", quantite=120, prix_unitaire_ht=45.0, total_ht=5400.0), LigneFacture(description="Faux plafond suspendu acoustique", quantite=80, prix_unitaire_ht=60.0, total_ht=4800.0) ], retenue_garantie_pct=5.0, autoliquidation_tva=False, taux_tva=20.0 ), ClientBTP( id_client="CL-002", nom_entreprise="Eiffage Construction (Sous-traitance)", nom_contact="Marie Curie", email="m.curie@eiffage.fr", telephone="01 40 50 60 70", adresse="45 Avenue de France, 69002 Lyon", nom_chantier="Résidence Les Alpes - Gros Œuvre", lignes_prestation=[ LigneFacture(description="Coulage de dalle béton armé", quantite=1, prix_unitaire_ht=15000.0, total_ht=15000.0) ], retenue_garantie_pct=5.0, autoliquidation_tva=True, # Autoliquidation TVA BTP taux_tva=0.0 ) ] # Génération automatique du PDF pour chaque client print("Début de la génération des documents PDF...") for client in liste_clients: generer_pdf_client(client) print("Tous les PDF ont été créés dans le dossier 'pdf_clients/'.")
